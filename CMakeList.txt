cmake_minimum_required(VERSION 3.20)
project(MacEngine LANGUAGES CXX OBJCXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 1. A motor könyvtár, ami tartalmazza a natív .mm fájlokat
add_library(MacEngine STATIC
    src/Platform/Mac/Window.mm
)

# A publikus fejlécek elérhetősége
target_include_directories(MacEngine PUBLIC include)

# 2. A kritikus rész: A Cocoa framework csatolása az Apple API-khoz
find_library(COCOA_LIBRARY Cocoa REQUIRED)
target_link_libraries(MacEngine PRIVATE ${COCOA_LIBRARY})

# 3. A teszt Sandbox alkalmazás
add_executable(Sandbox
    examples/main.cpp
)

# A Sandbox használja a motorunkat
target_link_libraries(Sandbox PRIVATE MacEngine)
